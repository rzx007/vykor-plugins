function parse(text) {
  const trimmed = text
    .replace(/\r?\n[0-9]+ passages from[\s\S]*$/, '')
    .replace(/\r?\nCite file:line for anything you rely on\.[\s\S]*$/, '');
  const candidates = codeCandidates(trimmed);
  const headers = [...trimmed.matchAll(
    /^─── .+?  \([^\n]*\)\r?\n(?:    \[[^\n]*?@ line (\d+)[^\n]*\](?:\r?\n|$))?/gm
  )];
  return candidates.map((candidate, index) => ({
    ...candidate,
    line: headers[index]?.[1] ? Number(headers[index][1]) : candidate.line,
    text: candidate.text
      .replace(/^    \[[^\n]*\]\s*$/, '')
      .replace(/\n    \.\.\. \[[^\n]* truncated at \d+ chars\]\s*$/, '')
      .trimEnd(),
  }));
}
